get_var_dict <- function(year) {
  if (length(year) != 1 || !(year %in% 2021:2024)) {
    stop("Year must be a single value between 2021 and 2024")
  }
  
  url <- paste0("https://api.census.gov/data/", year, "/acs/acs1/pums/variables.json?key=", key)
  response <- GET(url)
  stop_for_status(response)
  parsed <- fromJSON(content(response, as = "text", encoding = "UTF-8"), # Can also use the `rawToChar()` approach here 
                             simplifyVector = FALSE)
  
  state_var <- if (year <= 2022) "ST" else "STATE"
  
  #For each variable, keep a named character vector and sort by numeric order
  lapply(parsed$variables[c(vars, state_var)], function(vector) {
    items <- unlist(vector$values$item)
    code_num <- suppressWarnings(as.numeric(names(items)))
    items[order(is.na(code_num), code_num)]
  })
}

response_to_tibble <- function(response) {
  stop_for_status(response)
  parsed <- fromJSON(content(response, as = "text", encoding = "UTF-8"))
  colnames(parsed) <- parsed[1, ]
  as_tibble(parsed[-1, , drop = FALSE]) # drop = FALSE keeps matrix shape for the tibble
}

Then, let’s make some more helper functions to help us with our next step. These will convert variables into numeric values, normalize time codes and return the midpoint, and handle categorical variables by making them factors.

# Strip leading zeros to help standard time codes: "01", "1", and "001" all match
normalize_code <- function(x) {
   sub("^0+(?=.)", "", x, perl = TRUE) }

# Convert to numbers; codes labeled "N/A" in the dictionary become NA
convert_numeric <- function(x, labels) {
  na_codes <- normalize_code(names(labels)[grepl("^N/A", labels)])
  x_num <- suppressWarnings(as.numeric(x))
  x_num[normalize_code(x) %in% na_codes] <- NA
  x_num
}

# Time: midpoint of intervals as a time of day
convert_time <- function(x, labels) {
  clean <- gsub("\\.", "", labels)   # "a.m." -> "am"
  start <- parse_date_time(sub(" to .*", "", clean), "I:M p", quiet = TRUE)
  end   <- parse_date_time(sub(".* to ", "", clean), "I:M p", quiet = TRUE)
  mid_sec <- (hour(start) + hour(end)) * 1800 + (minute(start) + minute(end)) * 30
  hms(seconds = mid_sec[match(normalize_code(x), normalize_code(names(labels)))])
}

# Categorical: factor with dictionary-ordered levels and readable labels
make_factor <- function(x, labels) {
  factor(normalize_code(x), levels = normalize_code(names(labels)), labels = labels)
}

Now, using our helper functions, we’ll wrap this into a function that also parses the data by writing a function to query the API that allows the user to change the following items:

  • Year of survey (2024 as default)
    • Check a valid value given
  • Specify numeric variables to be returned (AGEP and PWGTP as default)
    • Turn variables into numeric values or time (use lubridate)
  • Specify the categorical variables to be returned (SEX as default)
    • Must include at least one categorical variable
    • Turn variables into factors with appropriate levels
  • Specify geography level: Region, Division, or State
  • Specify the specific region, division, or state to subset data on
    • Specify a default level if the user doesn’t
    • Allowable values are in the .rds file
    • Allow input all option that would return all values for that geography
  • Return an error if user specifies a variable not included in the above two lists
query_pums <- function(year = 2024,
                       num_vars = c("AGEP", "PWGTP"),
                       cat_vars = "SEX",
                       geo_level = "State",
                       geo_value = NULL,
                       api_key = key) {
  # Stop if invalid year
  if (length(year) != 1 || !(year %in% 2021:2024)) {
    stop("year must be a single value: 2021, 2022, 2023, or 2024")
  }
  # Numeric variable options
  num_options <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP")
  num_vars <- toupper(num_vars)
  bad_num <- setdiff(num_vars, c(num_options, "PWGTP"))
  if (length(bad_num) > 0) {
    stop("Invalid numeric variable(s): ", paste(bad_num, collapse = ", "), # Stop if invalid numeric variables requested
         ". Choose from: ", paste(num_options, collapse = ", "))
  }
  if (!any(num_vars %in% num_options)) {
    stop("Must request at least one numeric variable (other than PWGTP)") #Stop if no numeric variables requested
  }
  num_vars <- union(num_vars, "PWGTP")  # PWGTP is always returned
  # Categorical variable options
  cat_options <- c("FER", "HHL", "SCH", "SCHL", "SEX")
  cat_vars <- toupper(cat_vars)
  if (length(cat_vars) == 0) {
    stop("At least one categorical variable must be requested")
  }
  bad_cat <- setdiff(cat_vars, cat_options)
  if (length(bad_cat) > 0) {
    stop("Invalid categorical variable(s): ", paste(bad_cat, collapse = ", "), # Stop if invalid categorical variables requested
         ". Choose from: ", paste(cat_options, collapse = ", "))
  }
  # Geography level specification
  if (length(geo_level) != 1 ||
      !(tolower(geo_level) %in% c("region", "division", "state"))) {
    stop("geo_level must be one of 'Region', 'Division', or 'State'") # Stop if no Region, Division, or State given
  }
  geo_level <- tolower(geo_level)

  # Geography value options
  var_dicts <- readRDS("data/var_dicts.rds")
  geo_var <- switch(geo_level,
                    region   = "REGION",
                    division = "DIVISION",
                    state    = if (year <= 2022) "ST" else "STATE")
  geo_labels  <- var_dicts[[as.character(year)]][[geo_var]]
  valid_codes <- names(geo_labels)[!grepl("Puerto Rico", geo_labels)]

  # Defaults if no value given (chosen): South region (3), South Atlantic division (5), North Carolina (37)
  if (is.null(geo_value)) {
    geo_value <- switch(geo_level, region = "3", division = "5", state = "37")
  }

  if (length(geo_value) == 1 && tolower(geo_value) == "all") {
    geo_codes <- "*"
  } else {
    # Match integers so 1, "1", and "01" all work
    match_idx <- match(suppressWarnings(as.integer(geo_value)),
                       as.integer(valid_codes))
    if (anyNA(match_idx)) {
      stop("Invalid ", geo_level, " value(s): ", # Stop if invalid geo level
           paste(geo_value[is.na(match_idx)], collapse = ", "))
    }
    geo_codes <- paste(valid_codes[match_idx], collapse = ",")
  }
  # Query API and return a census tibble
  response <- GET(
    paste0("https://api.census.gov/data/", year, "/acs/acs1/pums"),
    query = list(get   = paste(c(num_vars, cat_vars), collapse = ","),
                 `for` = paste0(geo_level, ":", geo_codes),
                 key   = api_key)
  )
  pums_tbl <- response_to_tibble(response)
  
  # Convert column types using this years dictionary
  dict <- var_dicts[[as.character(year)]]
  for (v in num_vars) {
    pums_tbl[[v]] <- if (v %in% c("JWAP", "JWDP")) {
      convert_time(pums_tbl[[v]], dict[[v]])
    } else {
      convert_numeric(pums_tbl[[v]], dict[[v]])
    }
  }
  for (v in cat_vars) {
    pums_tbl[[v]] <- make_factor(pums_tbl[[v]], dict[[v]])
  }
  pums_tbl[[geo_level]] <- make_factor(pums_tbl[[geo_level]], dict[[geo_var]])

  class(pums_tbl) <- c("census", class(pums_tbl))
  pums_tbl
}

Combining Years

Finally, lets write a function that allows the user to specify multiple years of survey data (along with all the other options). This function will call a single year function as many times as needed and then combine the data into one final tibble with a year variable included.

query_pums_combined <- function(years = 2024, ...) { # ... passes everything else to query_pums (not repeated)
  years <- sort(unique(years))
  if (length(years) == 0 || !all(years %in% 2021:2024)) {
    stop("Years must be one or more of: 2021, 2022, 2023, 2024")
  }

  # Query each year, add a year column, and drop the census class before binding
  results <- lapply(years, function(yr) {
    year_tbl <- query_pums(year = yr, ...)
    year_tbl$year <- yr
    class(year_tbl) <- setdiff(class(year_tbl), "census")
    year_tbl
  })

  combined <- bind_rows(results)
  combined$year <- factor(combined$year)

  class(combined) <- c("census", class(combined))
  combined
}

Part 2: Writing a Generic Function for Summarizing

The rows in our tibbles don’t generally correspond to a single person! The PWGTP variable actually represents the number of (people) observations there should be for a particular row.

We could create a new tibble with each row replicated PWGTP times, but this would create a very large tibble in some cases.

Instead, we’ll use Generic Functions

Since we have added a census class to our tibble, we can write our own custom summary() function!

Numeric Summaries

For the census summary method, let’s write a function that produces means and standard deviations for our numeric variable(s) and counts for our categorical variable(s)

  • The function will take three arguments: the census tibble (maybe having multiple years), the numeric variable(s) to summarize, and the categorical variable(s) to summarize.

  • By default, it will summarize all numeric variables (other than PWGTP) and all categorical variables in the tibble. However, the user will be able to specify the variables they’d like to summarize.

  • All values will be returned as a named list

summary.census <- function(object, num_vars = NULL, cat_vars = NULL, ...) {
  weight <- object$PWGTP

  # Columns that are numeric but not time values
  is_plain_numeric <- function(col) is.numeric(col) && !inherits(col, "hms")

  # Defaults: all non-time numeric variables (except PWGTP) and all factors
  if (is.null(num_vars)) {
    num_vars <- names(object)[sapply(object, is_plain_numeric)]
    num_vars <- setdiff(num_vars, "PWGTP")
  }
  if (is.null(cat_vars)) {
    cat_vars <- names(object)[sapply(object, is.factor)]
  }

  # Validate user input
  missing_vars <- setdiff(c(num_vars, cat_vars), names(object))
  if (length(missing_vars) > 0) {
    stop("Variable(s) not in the data: ", paste(missing_vars, collapse = ", "))
  }
  if (!all(sapply(object[num_vars], is_plain_numeric))) {
    stop("num_vars must be numeric, non-time variables")
  }
  if (!all(sapply(object[cat_vars], is.factor))) {
    stop("cat_vars must be categorical (factor) variables")
  }

  # Weighted mean and SD for each numeric variable, dropping NAs
  num_summaries <- lapply(num_vars, function(v) {
    keep <- !is.na(object[[v]])
    x <- object[[v]][keep]
    w <- weight[keep]
    mean_x <- sum(x * w) / sum(w)
    sd_x <- sqrt(sum(x^2 * w) / sum(w) - mean_x^2)
    c(mean = mean_x, sd = sd_x)
  })
  names(num_summaries) <- num_vars

  # Weighted counts for each categorical variable
  cat_summaries <- lapply(cat_vars, function(v) {
    tapply(weight, object[[v]], sum, default = 0)
  })
  names(cat_summaries) <- cat_vars

  list(numeric = num_summaries, categorical = cat_summaries)
}

Graphical Summaries

Similar to this process, let’s create a generic plot() function for a census class tibble. This will require the user to specify one categorical variable (can include year) and one numeric variable for plotting. The plot should create a boxplot of the numeric variable at the levels of the categorical variable.

plot.census <- function(x, cat_var, num_var, ...) {
  # Both variables are required
  if (missing(cat_var) || missing(num_var)) {
    stop("You must specify one categorical (cat_var) and one numeric (num_var) variable")
  }
  if (length(cat_var) != 1 || length(num_var) != 1) {
    stop("cat_var and num_var must each be a single variable name")
  }

  # Validate the categorical variable
  if (!(cat_var %in% names(x)) || !is.factor(x[[cat_var]])) {
    stop(cat_var, " is not a categorical (factor) variable in the data")
  }

  # Validate the numeric variable (time values are allowed)
  if (!(num_var %in% names(x)) ||
      !(is.numeric(x[[num_var]]) || inherits(x[[num_var]], "hms"))) {
    stop(num_var, " is not a numeric variable in the data")
  }
  if (num_var == "PWGTP") {
    stop("PWGTP is the weight variable; choose a different numeric variable")
  }

  # Drop rows with missing values so ggplot doesn't warn about them
  plot_data <- x[!is.na(x[[num_var]]) & !is.na(x[[cat_var]]), ]

  ggplot(plot_data, aes(x = get(cat_var), y = get(num_var), weight = PWGTP)) +
    geom_boxplot() +
    labs(x = cat_var,
         y = num_var,
         title = paste("Plot of", num_var, "by", cat_var))
}
